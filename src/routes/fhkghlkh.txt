-- FUNCTION: Reduce stock from products table when a sale is made
CREATE OR REPLACE FUNCTION reduce_stock_after_sale()
RETURNS TRIGGER AS $$
BEGIN
    -- Check if enough stock is available
    IF (SELECT stock FROM products WHERE id = NEW.product_id) < NEW.quantity THEN
        RAISE EXCEPTION 'Not enough stock available for product ID %', NEW.product_id;
    END IF;

    -- Reduce stock
    UPDATE products
    SET stock = stock - NEW.quantity
    WHERE id = NEW.product_id;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- TRIGGER: Call the function after inserting into sales_items
CREATE TRIGGER trigger_reduce_stock
AFTER INSERT ON sales_items
FOR EACH ROW
EXECUTE FUNCTION reduce_stock_after_sale();
